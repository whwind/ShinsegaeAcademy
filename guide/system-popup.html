System Popup

Alert

Javascript CODE

/**
 * 메세지 팝업 함수
 * @param {String} type - 메세지 팝업 유형 (예시 : 'alert', 'confirm')
 * @param {String} title - 메세지 팝업 타이틀
 * @param {String} msg - 메세지 팝업 내용
 * @param {function} confirmCallback - 확인 버튼 클릭 후 실행할 콜백 함수
 * @param {function} cancelCallback - 취소 버튼 클릭 후 실행할 콜백 함수
 */

document.querySelector('.alert').addEventListener('click', () => {
	msgPopup({
		msg: '해당 게시물 상태를 검토로 변경 하시겠습니까?',
		confirmCallback: () => {
			alert('callback');
		},
	});
});

Confirm

Javascript CODE

/**
 * 메세지 팝업 함수
 * @param {String} type - 메세지 팝업 유형 (예시 : 'alert', 'confirm')
 * @param {String} title - 메세지 팝업 타이틀
 * @param {String} msg - 메세지 팝업 내용
 * @param {function} confirmCallback - 확인 버튼 클릭 후 실행할 콜백 함수
 * @param {function} cancelCallback - 취소 버튼 클릭 후 실행할 콜백 함수
 */

document.querySelector('.confirm').addEventListener('click', () => {
	msgPopup({
		type: 'confirm',
		title: '리소스신청이완료되었습니다.',
		msg: '신청된 내용은 신청목록 게시판을 통해 진행상태 및 결과를 안내합니다.',
		confirmCallback: () => {
			alert('callback');
		},
	});
});
