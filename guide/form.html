	type: 'text',                                 // Input Type  
	placeholder: 'Placeholder',                   // placeholder  
	size: '280px',                                // Width  
	class: 'is-datepicker end-date',              // Add Class  
	readonly: true,                               // readonly  
	disabled: true                                // disabled  
}) }}
Javascript CODE

const startInput = document.querySelector('.start-date');
const endInput = document.querySelector('.end-date');

const startDatepicker = new Datepicker(startInput, {
	autohide: true,
	format: 'yyyy-mm-dd'
});

const endDatepicker = new Datepicker(endInput, {
	autohide: true,
	format: 'yyyy-mm-dd'
});

// 시작일 선택 시 종료일의 최소 날짜를 설정
startInput.addEventListener('changeDate', (e) => {
	const startDate = startDatepicker.getDate();
	endDatepicker.setOptions({
		minDate: startDate
	});
});

// 종료일 선택 시 시작일의 최대 날짜를 설정
endInput.addEventListener('changeDate', (e) => {
	const endDate = endDatepicker.getDate();
	startDatepicker.setOptions({
		maxDate: endDate
	});
});

Design Select

Default Design Select

HTML CODE
Nunjucks CODE

{{ components('design-select', {
	id: 'c1',            					// ID  
	readonly: true,      					// readonly  
	disabled: true       					// disabled  
	checked: true       					// checked
	size: '280px'       					// Width  
	defaultText: '선택해주세요.'	// Default Text
	options: [          					// options
		{
			value: 's1',    					// option value
			text: '선택1' 					  // option text
		}
	]           
}) }}
Javascript CODE

/**
 * 디자인 셀렉트 박스 함수
 * @param {String} selector - 셀렉트 박스 선택자
 * @param {String} viewTarget - 옵션 내부 표시용 선택자 (예: ".design-select-option-text")
 * @param {Number} defaultIndex - 초기 선택할 옵션 인덱스 (0부터 시작)
 * @param {Function} afterEvent - 이벤트 이후 실행 함수 (e, value)
 */

selectBoxs({
	selector: '.design-select',
	afterEvent: (e, value) => {
		console.log('선택된 값:', value);
	}
});

/**
 * 셀렉트 박스의 선택값을 인덱스로 변경하는 함수
 * @param {string} selector - 셀렉트 박스 최상위 요소 선택자 (예: '.design-select')
 * @param {number} index - 선택할 옵션의 인덱스 (0부터 시작)
 * @param {function} [afterEvent] - 선택 후 호출할 콜백 함수 (e, value)
 */

setSelectValueByIndex('.design-select', 1, (e, value) => {
	alert('선택된 값: ' + value); // 문자열 + 변수
});

/**
 * 셀렉트 박스에서 현재 선택된 옵션의 인덱스를 반환
 * @param {string} selector - 셀렉트 박스 최상위 요소 선택자
 * @returns {number} 선택된 옵션 인덱스, 없으면 -1 반환
 */

 const selectedIndex = getSelectedIndex('.design-select');

 /**
 * 셀렉트 박스에서 현재 선택된 옵션의 data-value 값을 반환
 * @param {string} selector - 셀렉트 박스 최상위 요소 선택자
 * @returns {string|null} 선택된 옵션의 data-value, 없으면 null 반환
 */

const selectedValue = getSelectValue('.design-select');
